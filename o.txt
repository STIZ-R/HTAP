    min(_version) AS first_ts,
    max(_version) AS last_ts
FROM users;

SELECT CAST(now64(3), 'Float64') * 1000 AS ch_now_ms;

SELECT
    (CAST(now64(3), 'Float64')*1000 - min(_version)) AS lag_first_ms,
    (CAST(now64(3), 'Float64')*1000 - max(_version)) AS lag_last_ms
FROM users;

9)  check le nombre de parts dans clickhouse (Le + de parts, le - bien)
SELECT
    `table`,
    count() AS parts,
    sum(rows) AS rows,
    (sum(bytes_on_disk) / 1024) / 1024 AS mb,
    engine
FROM system.parts
WHERE `table` = 'users'
GROUP BY
    `table`,
    engine
ORDER BY parts DESC
